3. {% trans "Pool Rows" %}

{% trans "Pool rows are unsubmitted expense lines owned by the authenticated user. They can later be bundled into a draft expense application from the web UI." %}

{% trans "Method" %}{% trans "Endpoint" %}{% trans "Description" %}
GET /api/v1/pool/ {% trans "List the caller's pool rows, newest first." %}
POST /api/v1/pool/ {% trans "Create a pool row." %}
GET /api/v1/pool/<id>/ {% trans "Retrieve one pool row owned by the caller." %}
PATCH /api/v1/pool/<id>/ {% trans "Partially update a pool row owned by the caller." %}
PUT /api/v1/pool/<id>/ {% trans "Replace a pool row owned by the caller." %}
DELETE /api/v1/pool/<id>/ {% trans "Delete a pool row owned by the caller." %}

{% trans "Pool row fields" %}

{% trans "Field" %}{% trans "Required" %}{% trans "Notes" %}
id{% trans "No" %}{% trans "Read-only." %}
description{% trans "Yes" %}{% trans "Text description of the expense row." %}
begin_at{% trans "Yes" %}{% trans "ISO 8601 datetime, for example" %} 2026-09-04T06:00:00Z.
ended_at{% trans "Conditional" %}{% trans "Required when the selected expense type has" %} requires_endtime=true.
organisation{% trans "Yes" %}{% trans "Must be active and accessible to the user." %}
expensetype{% trans "Yes" %}{% trans "Must be active, accessible to the user, and belong to the same organisation." %}
accountdimension{% trans "No" %}{% trans "If supplied, it must belong to the same organisation. There is currently no API endpoint for discovering account dimensions." %}
basis{% trans "Yes" %}{% trans "Decimal amount, kilometres, days, or pieces depending on the expense type unit." %}
receipt{% trans "Conditional" %}{% trans "Required when the selected expense type has" %} requires_receipt=true. {% trans "Accepted file types are validated by the server." %}
expense{% trans "No" %}{% trans "Read-only. Pool rows always have" %} expense=null.
user{% trans "No" %}{% trans "Read-only. Set from the token owner." %}

{% trans "JSON create example" %}

POST /api/v1/pool/
Authorization: Token 0123456789abcdef...
Content-Type: application/json

{
  "description": "Taxi from station to meeting",
  "begin_at": "2026-09-04T12:00:00Z",
  "expensetype": 123,
  "basis": "25.50",
  "organisation": 45
}

{% trans "Multipart receipt upload example" %}

curl -X POST http://yhrek.docker.so/api/v1/pool/ \
  -H "Authorization: Token 0123456789abcdef..." \
  -F "description=Receipt upload test" \
  -F "begin_at=2026-09-04T12:00:00Z" \
  -F "expensetype=123" \
  -F "basis=25.50" \
  -F "organisation=45" \
  -F "receipt=@receipt.pdf"
